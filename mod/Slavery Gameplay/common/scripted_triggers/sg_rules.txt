sg_local_eligible = {
 total_size > 0
 NOT = { is_pop_type = slaves }
 pop_acceptance < acceptance_status_4
 wealth < 10
}
sg_seller_eligible = {
	OR = {
		is_country_type = decentralized
		AND = {
			is_country_type = unrecognized
			OR = {
				has_law_or_variant = law_type:law_debt_slavery
				sg_colonial_law = yes
				has_law_or_variant = law_type:law_slave_trade
				has_law_or_variant = law_type:law_legacy_slavery
			}
		}
		AND = {
			OR = {
				is_country_type = recognized
				is_country_type = colonial
				is_country_type = company
			}
			OR = {
				has_law_or_variant = law_type:law_debt_slavery
				sg_colonial_law = yes
				has_law_or_variant = law_type:law_slave_trade
			}
		}
	}
}
sg_slavery_law = {
 OR = {
  has_law_or_variant = law_type:law_debt_slavery
  sg_colonial_law = yes
  has_law_or_variant = law_type:law_slave_trade
  has_law_or_variant = law_type:law_legacy_slavery
 }
}
sg_colonial_law = {
 has_law_or_variant = law_type:law_colonial_slavery
}
sg_trade_source_eligible = {
 NOT = { owner = scope:sg_trade_controller }
 owner = { sg_seller_eligible = yes }
 OR = {
  any_scope_pop = { is_pop_type = slaves total_size > 0 }
  AND = { owner = { is_country_type = decentralized } any_scope_pop = { NOT = { is_pop_type = slaves } total_size > 0 } }
 }
}
